# Project 1 — Weather App

**Goal:** Build a weather app using the OpenWeather API with search, loading states, and error handling.

Get a free API key at \[openweathermap.org\](<https://openweathermap.org/api>&gt;). Free tier allows 60 calls/minute.

## Requirements

- \[ \] Search input for city name
- \[ \] Display: city name, country, temperature, description, humidity, wind
- \[ \] Weather icon from OpenWeather
- \[ \] Loading spinner while fetching
- \[ \] Error message for invalid city
- \[ \] 5-day forecast (bonus)

## Full Source Code

### script.js

```js
const API_KEY = 'YOUR_API_KEY_HERE'; // Get from openweathermap.org
const BASE = 'https://api.openweathermap.org/data/2.5';

// DOM elements
const form      = document.querySelector('#search-form');
const input     = document.querySelector('#city-input');
const loading   = document.querySelector('#loading');
const errorEl   = document.querySelector('#error');
const weatherEl = document.querySelector('#weather');

// Weather icons map
const weatherIcons = {
  Clear: '☀️', Clouds: '☁️', Rain: '🌧️',
  Drizzle: '🌦️', Thunderstorm: '⛈️', Snow: '❄️',
  Mist: '🌫️', Fog: '🌫️',
};

function showLoading()  { loading.hidden = false; weatherEl.hidden = true; errorEl.hidden = true; }
function showError(msg) { loading.hidden = true;  errorEl.textContent = msg; errorEl.hidden = false; }
function showWeather()  { loading.hidden = true;  weatherEl.hidden = false; }

async function fetchWeather(city) {
  showLoading();

  try {
    const [currentRes, forecastRes] = await Promise.all([
      fetch(`${BASE}/weather?q=${encodeURIComponent(city)}&appid=${API_KEY}&units=metric`),
      fetch(`${BASE}/forecast?q=${encodeURIComponent(city)}&appid=${API_KEY}&units=metric&cnt=40`),
    ]);

    if (!currentRes.ok) throw new Error(`City "${city}" not found`);

    const current  = await currentRes.json();
    const forecast = await forecastRes.json();

    renderCurrent(current);
    renderForecast(forecast);
    showWeather();
  } catch (err) {
    showError(err.message);
  }
}

function renderCurrent(data) {
  const icon = weatherIcons[data.weather[0].main] || '🌡️';
  document.querySelector('#weather-main').innerHTML = `
    <div class="city-name">${data.name}, ${data.sys.country}</div>
    <div class="weather-icon">${icon}</div>
    <div class="temp">${Math.round(data.main.temp)}°C</div>
    <div class="desc">${data.weather[0].description}</div>
    <div class="details">
      <span>💧 ${data.main.humidity}%</span>
      <span>🌬️ ${Math.round(data.wind.speed)} m/s</span>
      <span>👁️ ${(data.visibility / 1000).toFixed(1)} km</span>
      <span>🌡️ Feels ${Math.round(data.main.feels_like)}°C</span>
    </div>
  `;
}

function renderForecast(data) {
  // Get one entry per day (noon readings)
  const daily = data.list.filter(item => item.dt_txt.includes('12:00:00')).slice(0, 5);
  document.querySelector('#forecast').innerHTML = daily.map(day => {
    const date = new Date(day.dt * 1000).toLocaleDateString('en', { weekday: 'short', month: 'short', day: 'numeric' });
    const icon = weatherIcons[day.weather[0].main] || '🌡️';
    return `
      <div class="forecast-day">
        <div class="forecast-date">${date}</div>
        <div class="forecast-icon">${icon}</div>
        <div class="forecast-temp">${Math.round(day.main.temp)}°C</div>
      </div>
    `;
  }).join('');
}

form.addEventListener('submit', (e) => {
  e.preventDefault();
  const city = input.value.trim();
  if (city) fetchWeather(city);
});

// Load default city on start
fetchWeather('Yangon');
```
