# Project 2 — GitHub API Viewer

**Goal:** Search any GitHub user and display their profile + repositories using the GitHub REST API.

## Key API Endpoints

```
GET https://api.github.com/users/{username}
GET https://api.github.com/users/{username}/repos?sort=stars&per_page=6
```

## Core JavaScript

```js
const GITHUB_API = 'https://api.github.com';

async function fetchGitHubUser(username) {
  showLoading();
  try {
    const [userRes, reposRes] = await Promise.all([
      fetch(`${GITHUB_API}/users/${username}`),
      fetch(`${GITHUB_API}/users/${username}/repos?sort=stars&per_page=6`),
    ]);

    if (!userRes.ok) throw new Error(`User "${username}" not found`);

    const user  = await userRes.json();
    const repos = await reposRes.json();

    renderProfile(user);
    renderRepos(repos);
    showContent();
  } catch (err) {
    showError(err.message);
  }
}

function renderProfile(user) {
  document.querySelector('#profile').innerHTML = `
    <img src="${user.avatar_url}" alt="${user.login}" />
    <div class="profile-info">
      <h2>${user.name || user.login}</h2>
      <p class="username">@${user.login}</p>
      ${user.bio ? `<p class="bio">${user.bio}</p>` : ''}
      <div class="stats">
        <div><strong>${user.public_repos}</strong> Repos</div>
        <div><strong>${user.followers}</strong> Followers</div>
        <div><strong>${user.following}</strong> Following</div>
      </div>
      ${user.location ? `<p>📍 ${user.location}</p>` : ''}
      ${user.blog ? `<a href="${user.blog}" target="_blank">🔗 ${user.blog}</a>` : ''}
      <a href="${user.html_url}" target="_blank" class="btn">View on GitHub</a>
    </div>
  `;
}

function renderRepos(repos) {
  document.querySelector('#repos').innerHTML = repos.map(repo => `
    <a href="${repo.html_url}" target="_blank" class="repo-card">
      <h3>${repo.name}</h3>
      <p>${repo.description || 'No description'}</p>
      <div class="repo-stats">
        <span>⭐ ${repo.stargazers_count}</span>
        <span>🍴 ${repo.forks_count}</span>
        ${repo.language ? `<span class="lang">${repo.language}</span>` : ''}
      </div>
    </a>
  `).join('');
}
```
